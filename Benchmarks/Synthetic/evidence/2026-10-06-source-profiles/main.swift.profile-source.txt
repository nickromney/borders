import Foundation
import Darwin

// Profiling-only input. No BorderEngine, window enumeration, capture session,
// preferences, socket client, network, or hardware adapter is constructed.
@inline(never)
func selectSyntheticWindow(_ windows: [WindowCandidate]) -> Int {
    WindowSelection.primary(among: windows)?.order ?? -1
}

@inline(never)
func measureSyntheticLuminance(_ samples: [RGBSample]) -> Double {
    LuminanceMeasurement.mean(of: samples) ?? -1
}

let windows = (0..<1000).map {
    WindowCandidate(order: $0, frame: CGRect(x: 0, y: 0, width: 100 + $0, height: 100 + $0))
}
let samples = Array(repeating: RGBSample(red: 0.5, green: 0.5, blue: 0.5), count: 1600)
precondition(selectSyntheticWindow(windows) == 999)
precondition(abs(measureSyntheticLuminance(samples) - 0.5) < 1e-12)
if CommandLine.arguments.contains("--check") {
    print("Synthetic golden outputs passed; no timing baseline collected.")
    exit(0)
}
func argument(_ name: String) -> String? {
    guard let index = CommandLine.arguments.firstIndex(of: name) else { return nil }
    precondition(index + 1 < CommandLine.arguments.count, "Missing value for \(name)")
    return CommandLine.arguments[index + 1]
}
let selected = argument("--scenario")
precondition(selected == nil || ["focused-selection", "luminance-core"].contains(selected!))
let scenarios = selected.map { [$0] } ?? ["focused-selection", "luminance-core"]
if let duration = argument("--profile-seconds") {
    guard let seconds = Double(duration), seconds > 0, seconds <= 30, let scenario = selected else {
        fatalError("Profiling needs --scenario and duration in (0,30]")
    }
    let deadline = ProcessInfo.processInfo.systemUptime + seconds
    var checksum = 0.0
    var iterations = 0
    while ProcessInfo.processInfo.systemUptime < deadline {
        for _ in 0..<1000 {
            checksum += scenario == "focused-selection"
                ? Double(selectSyntheticWindow(windows)) : measureSyntheticLuminance(samples)
        }
        iterations += 1000
    }
    precondition(abs(checksum - Double(iterations) * (scenario == "focused-selection" ? 999 : 0.5)) < 1e-8)
    print("Synthetic profile \(scenario): \(iterations) operations, checksum \(checksum)")
    exit(0)
}
for scenario in scenarios {
    var runs: [Double] = []
    var checksum = 0.0
    for batch in 0..<23 {
        let start = ProcessInfo.processInfo.systemUptime
        for _ in 0..<1000 {
            checksum += scenario == "focused-selection"
                ? Double(selectSyntheticWindow(windows)) : measureSyntheticLuminance(samples)
        }
        let elapsed = (ProcessInfo.processInfo.systemUptime - start) * 1000
        if batch >= 3 { runs.append(elapsed) }
    }
    precondition(checksum == 23000 * (scenario == "focused-selection" ? 999 : 0.5))
    var usage = rusage()
    precondition(getrusage(RUSAGE_SELF, &usage) == 0)
    let output: [String: Any] = ["peak_rss_mib": Double(usage.ru_maxrss) / 1048576, "scenario": scenario, "iterations_per_run": 1000,
                               "discarded_batch_warmups": 3, "runs_ms": runs, "checksum": checksum]
    print(String(decoding: try JSONSerialization.data(withJSONObject: output, options: [.sortedKeys]), as: UTF8.self))
}
